### 1.3 后端实现

#### 权限中间件

**文件位置:** `/backend/src/middleware/permissions.ts`

```typescript
// 角色定义
export enum UserRole {
    ADMIN = 'admin',
    DEVELOPER = 'developer',
    EXTERNAL = 'external'
}

// 权限定义
export enum Permission {
    // User management
    VIEW_USERS = 'view_users',
    CREATE_USER = 'create_user',
    EDIT_USER = 'edit_user',
    DELETE_USER = 'delete_user',

    // Project management
    VIEW_PROJECTS = 'view_projects',
    CREATE_PROJECT = 'create_project',
    EDIT_PROJECT = 'edit_project',
    DELETE_PROJECT = 'delete_project',

    // ... 更多权限
}

// 角色-权限映射
const rolePermissions: Record<UserRole, Permission[]> = {
    [UserRole.ADMIN]: [/* 所有权限 */],
    [UserRole.DEVELOPER]: [/* 开发者权限 */],
    [UserRole.EXTERNAL]: [/* 外部成员权限 */]
};

// 权限检查中间件
export function requirePermission(...permissions: Permission[]) {
    return (req: Request, res: Response, next: NextFunction) => {
        const user = (req.session as any)?.user;
        if (!user) {
            return res.status(401).json({ message: 'Unauthorized' });
        }

        const hasPermission = permissions.some(permission =>
            hasPermission(user.role, permission)
        );

        if (!hasPermission) {
            return res.status(403).json({ message: 'Forbidden' });
        }

        next();
    };
}

// 仅认证检查(所有登录用户)
export function requireAuth(req: Request, res: Response, next: NextFunction) {
    const user = (req.session as any)?.user;
    if (!user) {
        return res.status(401).json({ message: 'Unauthorized' });
    }
    next();
}

// 阻止外部用户
export function blockExternal(req: Request, res: Response, next: NextFunction) {
    const user = (req.session as any)?.user;
    if (user.role === UserRole.EXTERNAL) {
        return res.status(403).json({ message: 'Forbidden' });
    }
    next();
}
```

### 1.4 前端实现

#### 权限Hook

**文件位置:** `/frontend/src/hooks/usePermissions.ts`

```typescript
export function usePermissions() {
    const { user, loading } = useCurrentUser();

    // 检查单个权限
    const hasPermission = (permission: Permission): boolean => {
        if (!user) return false;
        const permissions = rolePermissions[user.role] || [];
        return permissions.includes(permission);
    };

    // 检查多个权限(任一)
    const hasAnyPermission = (permissions: Permission[]): boolean => {
        return permissions.some(permission => hasPermission(permission));
    };

    // 检查多个权限(全部)
    const hasAllPermissions = (permissions: Permission[]): boolean => {
        return permissions.every(permission => hasPermission(permission));
    };

    // 角色检查
    const isAdmin = (): boolean => user?.role === UserRole.ADMIN;
    const isDeveloper = (): boolean => user?.role === UserRole.DEVELOPER;
    const isExternal = (): boolean => user?.role === UserRole.EXTERNAL;
    const canEdit = (): boolean => user?.role !== UserRole.EXTERNAL;

    return {
        user,
        loading,
        hasPermission,
        hasAnyPermission,
        hasAllPermissions,
        isAdmin,
        isDeveloper,
        isExternal,
        canEdit
    };
}
```

### 2.1 功能概述

仪表板提供系统的实时统计数据和可视化展示,帮助团队快速了解项目进展。

**核心数据模块:**

2. **任务分布** - 按状态分布的可视化饼图

### 2.3 SQL查询优化

Dashboard API使用多个优化的SQL查询:

```sql
-- 1. 概览统计(单次查询)
SELECT
    (SELECT COUNT(DISTINCT id) FROM projects) as total_projects,
    (SELECT COUNT(DISTINCT id) FROM stories) as total_stories,
    (SELECT COUNT(DISTINCT id) FROM tasks) as total_tasks,
    (SELECT COUNT(DISTINCT id) FROM sprints WHERE status = 'current') as active_sprints,
    (SELECT COUNT(DISTINCT id) FROM sprint_stories WHERE status = 'completed') as completed_stories,
    (SELECT COUNT(DISTINCT id) FROM sprint_tasks WHERE status = 'completed') as completed_tasks;

-- 2. 任务分布(当前迭代)
SELECT status, COUNT(*) as count
FROM sprint_tasks
WHERE sprint_id = (SELECT id FROM sprints WHERE status = 'current' LIMIT 1)
GROUP BY status;

-- 3. 迭代进度(最近4个迭代)
SELECT
    sp.sprint_number,
    sp.start_date,
    sp.end_date,
    COUNT(DISTINCT CASE WHEN ss.status = 'completed' THEN ss.story_id END) as completed_stories,
    COUNT(DISTINCT ss.story_id) as total_stories,
    COUNT(DISTINCT CASE WHEN st.status = 'completed' THEN st.task_id END) as completed_tasks,
    COUNT(DISTINCT st.task_id) as total_tasks
FROM sprints sp
LEFT JOIN sprint_stories ss ON sp.id = ss.sprint_id
LEFT JOIN sprint_tasks st ON sp.id = st.sprint_id
WHERE sp.id != -1
GROUP BY sp.id, sp.sprint_number, sp.start_date, sp.end_date
ORDER BY sp.start_date DESC
LIMIT 4;

-- 4. 团队绩效(TOP 5)
SELECT
    u.id,
    u.display_name,
    COUNT(DISTINCT CASE WHEN st.status = 'completed' THEN st.task_id END) as completed_tasks,
    COUNT(DISTINCT st.task_id) as total_tasks
FROM users u
LEFT JOIN sprint_tasks st ON u.id = st.assigned_to AND st.sprint_id = $1
WHERE u.role IN ('admin', 'developer')
GROUP BY u.id, u.display_name
HAVING COUNT(DISTINCT st.task_id) > 0
ORDER BY completed_tasks DESC
LIMIT 5;

-- 5. 最近动态(最新6条,Union Story和Task)
SELECT 'task' as type, t.id, t.title, st.status, st.updated_at, u.display_name as updated_by
FROM sprint_tasks st
JOIN tasks t ON st.task_id = t.id
LEFT JOIN users u ON st.assigned_to = u.id
WHERE st.sprint_id = $1
UNION ALL
SELECT 'story' as type, s.id, s.title, ss.status, ss.updated_at, u.display_name as updated_by
FROM sprint_stories ss
JOIN stories s ON ss.story_id = s.id
LEFT JOIN users u ON ss.assigned_to = u.id
WHERE ss.sprint_id = $1
ORDER BY updated_at DESC
LIMIT 6;
```

**使用场景:**
- Excel周报导出时显示需求来源
- 项目管理页面可编辑此字段
- 用于追踪项目来源和责任人

**更新后的Projects表结构:**
```sql
CREATE TABLE projects (
    id SERIAL PRIMARY KEY,
    software_name VARCHAR(255) NOT NULL,           -- 项目名称
    department_id INT REFERENCES departments(id),   -- 所属部门
    type_id INT REFERENCES project_types(id),       -- 项目类型
    source VARCHAR(255),                            -- 需求来源 ⭐新增
    priority VARCHAR(50) DEFAULT '中',              -- 优先级
    created_by INT REFERENCES users(id),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(software_name)
);
```

### 4.4 多迭代数据获取逻辑

Excel导出需要获取三个迭代的数据:

```typescript
interface SprintContext {
    lastSprint: Sprint;      // 上周迭代
    currentSprint: Sprint;   // 本周迭代
    nextSprint: Sprint;      // 下周迭代
}

// 获取迭代上下文
async function getSprintContext(currentSprintId: number): Promise<SprintContext> {
    // 按时间排序所有迭代
    const allSprints = await db.query(
        'SELECT * FROM sprints WHERE id != -1 ORDER BY start_date ASC'
    );

    const currentIndex = allSprints.findIndex(s => s.id === currentSprintId);

    return {
        lastSprint: allSprints[currentIndex - 1] || null,
        currentSprint: allSprints[currentIndex],
        nextSprint: allSprints[currentIndex + 1] || null
    };
}
```

**数据填充逻辑:**
- **上周总结**: 保留空白列,用户手动填写
- **本周任务**: 从`currentSprint`获取该用户的Task
- **本周进展**: 从`sprint_tasks`获取status和progress
- **下周计划**: 从`nextSprint`获取该用户的Task
- **完成率**: 计算 (已完成Story数 / 总Story数) * 100

## 5. Backlog迭代特殊处理

### 5.1 Backlog定义

**特殊迭代ID:** `-1`

**用途:**
- 存放尚未规划到具体迭代的Story和Task
- 作为"任务池",方便后续分配到正式迭代

### 5.2 Backlog限制

**前端限制:**
1. ❌ 不显示"导出周报"按钮
2. ❌ 不显示"激活迭代"按钮
3. ❌ 不允许删除
4. ❌ 不允许编辑

**实现代码:**
```tsx
// frontend/src/pages/SprintsPage.tsx
{sprint.id !== -1 && hasPermission(Permission.EXPORT_SUMMARY_REPORT) && (
    <Button onClick={handleExport}>导出周报</Button>
)}

{sprint.id !== -1 && hasPermission(Permission.ACTIVATE_SPRINT) && (
    <Button onClick={handleActivate}>激活</Button>
)}

{sprint.id !== -1 && hasPermission(Permission.DELETE_SPRINT) && (
    <Button onClick={handleDelete}>删除</Button>
)}
```

### 5.3 数据库初始化

**setup.sh添加Backlog创建:**
```sql
-- 创建Backlog迭代
INSERT INTO sprints (id, sprint_number, start_date, end_date, status)
VALUES (-1, 'Backlog', '2000-01-01', '2099-12-31', 'planned')
ON CONFLICT (id) DO NOTHING;
```
